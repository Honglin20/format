```python
# src/transform/pre_scale.py
"""PreScaleTransform: learnable per-channel pre-scale via the Transform slot."""
import torch
from torch import Tensor
from src.scheme.transform import TransformBase


class PreScaleTransform(TransformBase):
    """Pre-scale transform: x → x * scale, x_q → x_q / scale.

    Holds a **reference** (not a copy) to an externally-owned scale tensor.
    This allows the tensor to be an ``nn.Parameter`` updated by an optimizer,
    or a buffer written by calibration — the transform automatically sees
    the latest values on the next forward pass.

    invertible = True.
    """

    invertible = True

    def __init__(self, scale: Tensor):
        """Create PreScaleTransform with a reference to *scale*.

        Args:
            scale: Tensor to use as pre-scale. Typically per-channel
                   (shape ``[C]`` or ``[C, 1, 1]``). NOT cloned —
                   external mutations are visible on next forward.
        """
        if not isinstance(scale, torch.Tensor):
            raise TypeError(f"scale must be a torch.Tensor, got {type(scale).__name__}")
        object.__setattr__(self, "scale", scale)

    def forward(self, x: Tensor) -> Tensor:
        return x * self.scale

    def inverse(self, x_q: Tensor) -> Tensor:
        return x_q / self.scale

    def __eq__(self, other):
        if not isinstance(other, PreScaleTransform):
            return NotImplemented
        return self.scale is other.scale

    def __hash__(self):
        return hash(("PreScaleTransform", id(self.scale)))

    def __repr__(self):
        return f"PreScaleTransform(shape={tuple(self.scale.shape)})"
```

```python
def quantize_mx(
    A,
    scheme,
    axes=None,
    scale_bits=8,
    shared_exp_method="max",
    flush_fp32_subnorms=False,
):
    """Quantize tensor A using MX block quantization.

    When *scheme.transform* is not IdentityTransform, delegates to the
    standard three-step :func:`quantize` flow (ADR-001 compliant).

    When *scheme.transform* is IdentityTransform, uses the direct MX
    path (``_quantize_mx``) as a fast path.

    Args:
        A: Input tensor.
        scheme: QuantScheme. granularity.mode must be PER_BLOCK or PER_TENSOR.
        axes: Axes for shared exponent computation.
        scale_bits: Bits for shared scale (sign + magnitude). Default: 8.
        shared_exp_method: "max" or "none". Default: "max".
        flush_fp32_subnorms: Flush subnormal FP32 blocks to zero.

    Returns:
        Quantized tensor with same shape as A.

    Raises:
        ValueError: If scheme.granularity is PER_CHANNEL.
    """
    if scheme is None:
        return A

    from src.scheme.granularity import GranularityMode
    from src.scheme.transform import IdentityTransform

    # Delegate to ADR-001 three-step flow when transform is non-identity
    if not isinstance(scheme.transform, IdentityTransform):
        from src.quantize.elemwise import quantize
        return quantize(A, scheme)

    if scheme.granularity.mode == GranularityMode.PER_CHANNEL:
        raise ValueError(
            "quantize_mx does not support PER_CHANNEL granularity. "
            "Use quantize(x, scheme) for per-channel quantization."
        )

    fmt = scheme.format
    block_size = scheme.block_size
    round_mode = scheme.round_mode

    return _quantize_mx(
        A, scale_bits, fmt,
        block_size=block_size,
        axes=axes, round_mode=round_mode,
        shared_exp_method=shared_exp_method,
        flush_fp32_subnorms=flush_fp32_subnorms,
    )
```

```python
# src/calibration/lsq_optimizer.py
"""LayerwiseScaleOptimizer: gradient-based per-layer pre-scale optimization.

Path A (BRECQ-style): sequential optimization using real quantized intermediates.
For each layer, runs the partially-quantized model to get true inputs,
then optimizes pre-scale via gradient descent to minimize MSE against
fp32 layer output.
"""
import copy
from typing import Dict, List, Optional

import torch
import torch.nn as nn

from src.scheme.op_config import OpQuantConfig
from src.scheme.quant_scheme import QuantScheme
from src.transform.pre_scale import PreScaleTransform


def _get_quantized_modules(model: nn.Module) -> List[tuple]:
    """Return [(name, module), ...] for all Quantized* modules with cfg."""
    result = []
    for name, module in model.named_modules():
        if hasattr(module, "cfg") and not getattr(module, "_is_passthrough", False):
            result.append((name, module))
    return result


def _replace_transform(cfg: OpQuantConfig, transform) -> OpQuantConfig:
    """Return a new OpQuantConfig with *transform* replacing all non-None schemes."""
    fields = {}
    for f in cfg.__dataclass_fields__:
        old = getattr(cfg, f)
        if old is not None and isinstance(old, QuantScheme):
            fields[f] = QuantScheme(
                format=old.format,
                granularity=old.granularity,
                transform=transform,
                round_mode=old.round_mode,
            )
        else:
            fields[f] = old
    return OpQuantConfig(**fields)


class LayerwiseScaleOptimizer:
    """Gradient-based per-layer pre-scale optimization.

    Uses Path A (sequential quantized intermediates):
    1. Run partially-quantized model to get layer N's real input
    2. Get fp32 target from fp32 model
    3. Gradient-optimize layer N's pre_scale: minimize MSE(y_fp32, y_quant)
    4. Freeze layer N's pre_scale, continue to N+1

    Args:
        num_steps: Optimization steps per layer (default: 100).
        num_batches: Number of calibration batches to use (default: 8).
        optimizer: Optimizer name — "adam" or "sgd" (default: "adam").
        lr: Learning rate (default: 1e-3).
        loss: Loss function — "mse" (default: "mse").
    """

    _VALID_OPTIMIZERS = {"adam", "sgd"}
    _VALID_LOSSES = {"mse"}

    def __init__(
        self,
        num_steps: int = 100,
        num_batches: int = 8,
        optimizer: str = "adam",
        lr: float = 1e-3,
        loss: str = "mse",
    ):
        if num_steps < 1:
            raise ValueError(f"num_steps must be >= 1, got {num_steps}")
        if num_batches < 1:
            raise ValueError(f"num_batches must be >= 1, got {num_batches}")
        if lr <= 0:
            raise ValueError(f"lr must be > 0, got {lr}")
        if optimizer not in self._VALID_OPTIMIZERS:
            raise ValueError(
                f"optimizer must be one of {self._VALID_OPTIMIZERS}, got {optimizer!r}"
            )
        if loss not in self._VALID_LOSSES:
            raise ValueError(
                f"loss must be one of {self._VALID_LOSSES}, got {loss!r}"
            )

        self.num_steps = num_steps
        self.num_batches = num_batches
        self.optimizer = optimizer
        self.lr = lr
        self.loss = loss

    def optimize(
        self,
        qmodel: nn.Module,
        fp32_model: nn.Module,
        calib_batches: List[torch.Tensor],
    ) -> Dict[str, torch.Tensor]:
        """Run layer-wise LSQ optimization.

        Args:
            qmodel: Quantized model (from quantize_model).
            fp32_model: Original fp32 model.
            calib_batches: List of input tensors from calibration.

        Returns:
            Dict mapping module name → optimized pre_scale tensor.
        """
        modules = _get_quantized_modules(qmodel)
        batches = calib_batches[:self.num_batches]
        optimized_scales: Dict[str, torch.Tensor] = {}

        # Collect fp32 intermediates for all layers in one pass
        fp32_intermediates = self._collect_fp32_targets(
            qmodel, fp32_model, modules, batches
        )

        for layer_idx, (name, module) in enumerate(modules):
            # Find the channel count for this layer's output
            init_scale = self._initial_scale(module, batches, fp32_model, name)
            pre_scale = nn.Parameter(init_scale.clone())

            # Temporarily inject PreScaleTransform into module's cfg
            original_cfg = module.cfg
            transform = PreScaleTransform(scale=pre_scale)
            module.cfg = _replace_transform(original_cfg, transform)

            # Build optimizer
            if self.optimizer == "adam":
                opt = torch.optim.Adam([pre_scale], lr=self.lr)
            else:
                opt = torch.optim.SGD([pre_scale], lr=self.lr)

            loss_fn = nn.MSELoss()

            # Get the "real" inputs: run partially-quantized model
            # (layers 0..layer_idx quantized, rest passthrough)
            real_inputs = self._get_layer_inputs(
                qmodel, modules, layer_idx, name, batches
            )

            # Targets from fp32 model
            targets = fp32_intermediates[name]

            # Optimize
            module.train()
            for step in range(self.num_steps):
                total_loss = 0.0
                for x_in, y_fp32 in zip(real_inputs, targets):
                    opt.zero_grad()
                    y_q = module(x_in)
                    loss = loss_fn(y_q, y_fp32)
                    loss.backward()
                    opt.step()
                    total_loss += loss.item()
            module.eval()

            # Freeze: store optimized scale
            optimized_scale = pre_scale.detach()
            optimized_scales[name] = optimized_scale

            # Replace cfg transform with frozen PreScaleTransform
            frozen_transform = PreScaleTransform(scale=optimized_scale)
            module.cfg = _replace_transform(original_cfg, frozen_transform)

            # Persist as module buffer for inference
            module.register_buffer("_pre_scale", optimized_scale)

        return optimized_scales

    def _initial_scale(self, module, batches, fp32_model, name):
        """Compute initial pre-scale from calibration data (absmax per channel)."""
        # Run fp32 model to get layer output, use per-channel absmax as init
        all_outputs = []
        with torch.no_grad():
            for batch in batches[:self.num_batches]:
                out = fp32_model(batch)
                all_outputs.append(out)

        # Guess output channels from last dim of output
        output_shape = all_outputs[0].shape
        out_channels = output_shape[-1]

        # Compute per-channel absmax
        stacked = torch.cat([o.reshape(-1, out_channels) for o in all_outputs], dim=0)
        amax = torch.amax(torch.abs(stacked), dim=0).clamp(min=1e-12)
        return amax

    def _collect_fp32_targets(self, qmodel, fp32_model, modules, batches):
        """Collect fp32 layer outputs for all quantized layers."""
        targets = {}
        fp32_modules = dict(fp32_model.named_modules())

        with torch.no_grad():
            for batch in batches:
                # For each quantized layer, find the corresponding fp32 output
                x = batch
                for name, _ in modules:
                    fp32_mod = fp32_modules.get(name)
                    if fp32_mod is not None:
                        x = fp32_mod(x)
                        if name not in targets:
                            targets[name] = []
                        targets[name].append(x.clone())
        return targets

    def _get_layer_inputs(self, qmodel, modules, layer_idx, name, batches):
        """Get real inputs to layer N by running partially-quantized model."""
        inputs = []

        # Set layers before N to quantized, N..L-1 to passthrough
        saved_cfgs = {}
        for i, (mod_name, mod) in enumerate(modules):
            saved_cfgs[mod_name] = mod.cfg
            if i < layer_idx:
                # Already optimized — keep quantized cfg
                pass
            elif i == layer_idx:
                # Current layer — use quantized cfg (with PreScaleTransform)
                pass
            else:
                # After current layer — passthrough
                mod.cfg = OpQuantConfig()

        with torch.no_grad():
            for batch in batches[:self.num_batches]:
                x = batch
                # Run through all layers before N
                for i, (mod_name, mod) in enumerate(modules):
                    if i == layer_idx:
                        inputs.append(x.clone())
                        break
                    x = mod(x)

        # Restore cfgs
        for mod_name, mod in modules:
            mod.cfg = saved_cfgs[mod_name]

        return inputs
```

```python
    # ------------------------------------------------------------------
    # Pre-scale (P5: learnable pre-scale via Transform)
    # ------------------------------------------------------------------

    def initialize_pre_scales(
        self,
        calib_data: List[torch.Tensor],
        init: str = "absmax",
    ) -> int:
        """Initialize ``_pre_scale`` buffers on all quantized modules.

        Creates per-channel pre-scale tensors initialized from calibration
        data, registers them as buffers, and updates each module's cfg
        to route quantization through ``PreScaleTransform``.

        Args:
            calib_data: List of input tensors from calibration.
            init: Initialization method — ``"absmax"`` (per-channel absmax)
                  or ``"ones"`` (all ones).

        Returns:
            Number of modules that received pre-scale buffers.
        """
        from src.transform.pre_scale import PreScaleTransform

        count = 0
        for name, module in self.qmodel.named_modules():
            if not hasattr(module, "cfg") or getattr(module, "_is_passthrough", False):
                continue

            # Determine output channels for this module
            out_channels = self._infer_out_channels(module, calib_data)
            if out_channels is None:
                continue

            # Create initial pre-scale
            if init == "absmax":
                init_scale = self._compute_absmax_scale(module, calib_data, out_channels)
            elif init == "ones":
                init_scale = torch.ones(out_channels)
            else:
                raise ValueError(f"Unknown init method: {init!r}")

            # Register as buffer
            module.register_buffer("_pre_scale", init_scale)

            # Update cfg to route through PreScaleTransform
            transform = PreScaleTransform(scale=module._pre_scale)
            module.cfg = _replace_transforms_in_cfg(module.cfg, transform)
            count += 1

        return count

    def optimize_scales(
        self,
        optimizer: "LayerwiseScaleOptimizer",
        calib_data: List[torch.Tensor],
    ) -> Dict[str, torch.Tensor]:
        """Run layer-wise LSQ optimization on pre-scale parameters.

        Args:
            optimizer: Configured ``LayerwiseScaleOptimizer`` instance.
            calib_data: List of input tensors from calibration data.

        Returns:
            Dict mapping module name → optimized pre_scale tensor.

        Raises:
            RuntimeError: If ``fp32_model`` is not available
                (``keep_fp32=False`` was passed to QuantSession).
        """
        if self.fp32_model is None:
            raise RuntimeError(
                "optimize_scales requires fp32_model (keep_fp32=True)"
            )

        from src.calibration.lsq_optimizer import LayerwiseScaleOptimizer
        return optimizer.optimize(self.qmodel, self.fp32_model, calib_data)

    @staticmethod
    def _infer_out_channels(module, calib_data):
        """Infer output channel count for a module."""
        # Try common attributes
        if hasattr(module, "out_features"):
            return module.out_features
        if hasattr(module, "out_channels"):
            return module.out_channels
        if hasattr(module, "num_features"):
            return module.num_features
        # Fallback: run one forward
        try:
            with torch.no_grad():
                out = module(calib_data[0])
            return out.shape[-1]
        except Exception:
            return None

    @staticmethod
    def _compute_absmax_scale(module, calib_data, out_channels):
        """Compute per-channel absmax from calibration data.

        Fixes the format's internal scale — only pre_scale is learned.
        """
        all_outs = []
        with torch.no_grad():
            for batch in calib_data[:8]:
                out = module(batch)
                all_outs.append(out.reshape(-1, out_channels))

        stacked = torch.cat(all_outs, dim=0)
        amax = torch.amax(torch.abs(stacked), dim=0).clamp(min=1e-12)
        return amax
```

### Task 5: Internal Scale Fixing During LSQ

**Files:**
- Modify: `src/calibration/lsq_optimizer.py` (fix per-channel amax via `scale=` parameter)

**Step 1: Ensure per-channel internal scale is fixed during optimization**

In `LayerwiseScaleOptimizer.optimize()`, when a module's cfg has per-channel granularity, the internal amax must be fixed during LSQ. The `_replace_transform()` function already preserves the original scheme's format and granularity. The internal `scale=` parameter can be passed to `quantize()` via the module's forward.

This requires a small modification to how we configure the module during optimization: when the scheme uses PER_CHANNEL, we pre-compute and fix the internal amax.

Add a helper to the LSQ optimizer:

```python
def _fix_internal_scale(self, cfg: OpQuantConfig, module, calib_batches):
    """Pre-compute internal per-channel amax and fix it.

    For per-channel granularity, compute amax from calibration data
    and store it so the module's forward passes it as scale= to quantize().
    This ensures only pre_scale is optimized, not internal amax.
    """
    from src.scheme.granularity import GranularityMode

    new_fields = {}
    for f_name, scheme in cfg.__dataclass_fields__.items():
        old = getattr(cfg, f_name)
        if old is None or not isinstance(old, QuantScheme):
            new_fields[f_name] = old
            continue
        if old.granularity.mode != GranularityMode.PER_CHANNEL:
            new_fields[f_name] = old
            continue

        # Compute fixed amax from calibration
        all_outs = []
        with torch.no_grad():
            for batch in calib_batches:
                out = module(batch)
                all_outs.append(out)
        stacked = torch.cat([o.reshape(-1, o.shape[-1]) for o in all_outs], dim=0)
        amax = torch.amax(torch.abs(stacked), dim=0, keepdim=True).clamp(min=1e-12)

        # Store on module for forward to use
        module.register_buffer(f"_internal_scale_{f_name}", amax)
        new_fields[f_name] = old  # scheme unchanged, scale injected via buffer

    return OpQuantConfig(**new_fields)
```

This helper is called during `optimize()` for per-channel schemes. The module's forward already reads `_output_scale` from its buffer — we can extend that pattern.

Actually, looking at the code in `QuantizedLinear.forward()`:
```python
output_scale = self.get_buffer("_output_scale") \
    if hasattr(self, "_output_scale") else None
```

The existing pattern passes `output_scale` to `LinearFunction.apply()`, which passes it to `quantize()`. We need a similar pattern for the internal amax of per-channel weight/input quantization.

For P5, the simplest approach: when LSQ is active and scheme is PER_CHANNEL, pre-compute the per-channel amax from calibration data and store it as a buffer. The module's forward uses it as the fixed internal scale. Only pre_scale is learnable.

This step primarily ensures correctness — it makes the "internal scale fixed" principle from ADR-006 concrete.
